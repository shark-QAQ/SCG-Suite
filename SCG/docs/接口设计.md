这里是 **SCG (Smart Code Guard)** 的完整接口设计文档。

为了保证工程级的高内聚低耦合，我们将接口分为四大类：**1. DSH 扩展入口接口**（插件生命周期）、**2. 模型调用接口**（Tool 契约）、**3. Harness 拦截接口**（Hook 契约）、**4. 内部模块接口**（业务逻辑解耦）。

---

### 1. DSH 扩展入口接口 (Plugin Lifecycle)

这是插件与 DSH (Cordis 内核) 交互的边界。必须声明正确的依赖和清理逻辑。

```typescript
import type { Context } from '@deepseek-ai/cordis'

// 插件名称标识
export const name = 'scg'

// 声明依赖：必须在 tools 服务就绪后加载
export const inject = ['tools']

// 插件配置结构
export interface SCGConfig {
  enabled: boolean;          // 总开关，默认 true
  blockOnHigh: boolean;      // 是否拦截高危命令，默认 true
  extraRulesPath?: string;   // 自定义规则文件路径
}

// 插件主入口
export function apply(ctx: Context, config: SCGConfig) {
  if (!config.enabled) {
    ctx.logger.info('[SCG] 插件已禁用')
    return
  }

  // 1. 注册 Tool (使用 ctx.effect 管理生命周期)
  ctx.effect(() => {
    const disposeTool = ctx.tools.register(ScgScanTool)
    return () => disposeTool()
  })

  // 2. 注册 Hook 拦截器 (使用 ctx.effect 管理生命周期)
  ctx.effect(() => {
    const disposeHook = ctx.on('tools/pre-execute', ScgInterceptor)
    return () => disposeHook()
  })

  ctx.logger.info('[SCG] Smart Code Guard 加载完成')
}
```

### 2. 模型调用接口 (Tool Contract)

这是提供给 LLM 调用的工具契约。必须严格遵循 JSON Schema 规范，输出必须是扁平的 Canonical Value，避免模型解析困难。

#### 2.1 工具定义 (JSON Schema)

```typescript
export const ScgScanTool = {
  name: 'scg_scan',
  description: 'SCG 安全扫描：对一段代码片段进行静态检查，识别高危漏洞（如硬编码密钥、SQL 注入、危险命令）。',
  parameters: {
    type: 'object',
    properties: {
      code: { 
        type: 'string', 
        description: '需要扫描的代码内容' 
      },
      language: { 
        type: 'string', 
        enum: ['bash', 'sql', 'javascript', 'python', 'typescript', 'unknown'],
        description: '代码的编程语言，用于匹配特定语言的规则。默认为 unknown。'
      }
    },
    required: ['code']
  }
}
```

#### 2.2 输入输出类型定义 (TypeScript)

```typescript
// 工具输入参数类型
export interface ScgScanInput {
  code: string;
  language?: 'bash' | 'sql' | 'javascript' | 'python' | 'typescript' | 'unknown';
}

// 单个问题项
export interface ScgIssue {
  ruleId: string;                           // 规则 ID，如 'SCG-001'
  severity: 'high' | 'medium' | 'low';      // 严重级别
  message: string;                          // 错误描述
  line?: number;                            // 命中行号（能算则算）
  suggestion: string;                       // 修复建议
}

// 工具输出结果 (Canonical Value)
export interface ScgScanResult {
  safe: boolean;                            // 是否安全 (issues 为空则为 true)
  summary: string;                          // 一句话结论
  issues: ScgIssue[];                       // 问题列表
}
```

### 3. Harness 拦截接口 (Hook Contract)

这是 SCG 与 DSH Agent Loop 交互的边界。利用拦截器（Interceptor）模式，在工具真正执行前进行阻断。

```typescript
// DSH 提供的执行上下文 (简化示意)
export interface ToolExecution {
  name: string;                             // 工具名，如 'bash'
  arguments: Record<string, any>;           // 模型传入的参数，如 { command: 'rm -rf /' }
}

// DSH 期望的拦截决策返回
export type PreToolDecision = 
  | { kind: 'allow' } 
  | { kind: 'deny'; reason: string }
  | { kind: 'ask'; reason: string };        // 请求人类审批（企业版备选）

// SCG 拦截器实现
export async function ScgInterceptor(
  exec: ToolExecution, 
  next: () => Promise<PreToolDecision>
): Promise<PreToolDecision> {
  
  // 1. 白名单过滤：只拦截高危工具
  const monitoredTools = ['bash', 'execute_command', 'write_file'];
  if (!monitoredTools.includes(exec.name)) {
    return next(); // 直接放行
  }

  // 2. 针对 bash 命令进行危险检测
  if (exec.name === 'bash' || exec.name === 'execute_command') {
    const command = exec.arguments?.command || '';
    
    // 调用内部规则引擎
    const danger = evaluateDangerousCommand(command);
    
    if (danger.isDangerous) {
      return { 
        kind: 'deny', 
        reason: `🚨 [SCG 拦截] 该操作被安全策略阻止。原因: ${danger.reason}。请使用其他安全的方式完成任务。` 
      };
    }
  }

  // 3. 其他情况放行
  return next();
}
```

      config:
        enabled: true
        blockOnHigh: true
        # 可选：加载自定义规则扩展
